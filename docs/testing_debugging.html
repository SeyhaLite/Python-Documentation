Testing, Debugging និង Best Practices

ហេតុអ្វីត្រូវសាកល្បង?

អត្ថប្រយោជន៍

ការពន្យល់

រកកំហុសមុនអ្នកប្រើ

test ដែលដំណើរការស្វ័យប្រវត្តិរកឃើញ regression ភ្លាម

Refactor ដោយទំនុកចិត្ត

បើ test នៅឆ្លង អ្នកដឹងថាអាកប្បកិរិយាមិនបានប្រែប្រួល

ឯកសារដែលមានជីវិត

test បង្ហាញពីរបៀបប្រើ API — វាមិនអាចខុសពីកូដបានទេ

រចនាល្អជាង

កូដដែលងាយសាកល្បង ជាធម្មតាមាន coupling ទាប និងទំនួលខុសត្រូវតែមួយ

Test Pyramid

1    /\        ← End-to-end (តិច យឺត ថ្លៃ)
2   /  \
3  /----\      ← Integration (មធ្យម)
4 /      \
5/--------\    ← Unit (ច្រើន លឿន ថោក)

  • Unit test — សាកល្បងអនុគមន៍ ឬ class តែមួយ ដាច់ដោយឡែក។ សរសេរច្រើនបំផុត។

  • Integration test — សាកល្បងផ្នែកច្រើនធ្វើការជាមួយគ្នា (database, API, ឯកសារ)។

  • End-to-end test — សាកល្បងកម្មវិធីទាំងមូលពីមុខអ្នកប្រើ។ សរសេរតិចបំផុត ព្រោះយឺត និងងាយបែក។

doctest — សាកល្បងពីឧទាហរណ៍

doctest ស្វែងរកឧទាហរណ៍ដែលមានទម្រង់ REPL (>>>) ក្នុង docstring ហើយដំណើរការវា៖

 1def celsius_to_fahrenheit(celsius: float) -> float:
 2    """បំលែង Celsius ទៅ Fahrenheit។
 3
 4    >>> celsius_to_fahrenheit(0)
 5    32.0
 6    >>> celsius_to_fahrenheit(100)
 7    212.0
 8    >>> round(celsius_to_fahrenheit(-40), 1)
 9    -40.0
10    """
11    return celsius * 9 / 5 + 32

 1>>> import doctest
 2>>> doctest.run_docstring_examples(
 3...     celsius_to_fahrenheit,
 4...     {"celsius_to_fahrenheit": celsius_to_fahrenheit},
 5...     verbose=False,
 6... )
 7>>> celsius_to_fahrenheit(0)
 832.0
 9>>> celsius_to_fahrenheit(100)
10212.0

Directive សំខាន់ៗ

Directive

ការប្រើប្រាស់

# doctest: +ELLIPSIS

អនុញ្ញាត ... ជា wildcard

# doctest: +SKIP

រំលងឧទាហរណ៍នេះ

# doctest: +NORMALIZE_WHITESPACE

មិនអើពើដកឃ្លាខុសគ្នា

# doctest: -IGNORE_EXCEPTION_DETAIL

ប្រៀបធៀបតែប្រភេទ exception

Tip

doctest ល្អសម្រាប់ ឯកសារដែលអាចផ្ទៀងផ្ទាត់បាន — ឧទាហរណ៍ក្នុង docstring របស់អ្នកនឹងមិនអាចខុសពីកូដបានទេ។ ប៉ុន្តែវា មិនគ្រប់គ្រាន់ សម្រាប់ការសាកល្បងពេញលេញទេ — ប្រើ unittest ឬ pytest ផង។

unittest

unittest គឺជា framework សាកល្បងស្តង់ដារដែលមកជាមួយ Python។

ឧទាហរណ៍ពេញលេញ

test_calculator.py
 1"""ឧទាហរណ៍ unit test សម្រាប់ម៉ាស៊ីនគិតលេខសាមញ្ញ។"""
 2
 3import unittest
 4
 5
 6def add(a, b):
 7    """បូកលេខពីរ។"""
 8    return a + b
 9
10
11def multiply(a, b):
12    """គុណលេខពីរ។"""
13    return a * b
14
15
16def divide(a, b):
17    """ចែកលេខពីរ។ បញ្ចេញ ZeroDivisionError បើ b == 0។"""
18    return a / b
19
20
21class TestCalculator(unittest.TestCase):
22    """សាកល្បងអនុគមន៍គណនា។"""
23
24    def setUp(self):
25        self.pairs = [(1, 2, 3), (0, 0, 0), (-1, 1, 0), (2.5, 0.5, 3.0)]
26
27    def test_add(self):
28        for a, b, expected in self.pairs:
29            with self.subTest(a=a, b=b):
30                self.assertEqual(add(a, b), expected)
31
32    def test_multiply(self):
33        self.assertEqual(multiply(3, 4), 12)
34        self.assertEqual(multiply(-2, 5), -10)
35        self.assertEqual(multiply(0, 100), 0)
36
37    def test_divide_by_zero(self):
38        with self.assertRaises(ZeroDivisionError):
39            divide(1, 0)
40
41    def test_divide(self):
42        self.assertAlmostEqual(divide(1, 3), 0.3333333, places=6)
43
44
45if __name__ == "__main__":
46    unittest.main(verbosity=2)

Method

ពិនិត្យ

assertEqual(a, b)

a == b

assertNotEqual(a, b)

a != b

assertTrue(x) / assertFalse(x)

bool(x) ពិត / មិនពិត

assertIs(a, b) / assertIsNot

a is b

assertIsNone(x) / assertIsNotNone

x is None

assertIn(a, b) / assertNotIn

a in b

assertIsInstance(a, b)

isinstance(a, b)

assertAlmostEqual(a, b, places=7)

exception ជាមួយសារដែលផ្គូផ្គង

assertWarns(Warning)

បញ្ចេញ warning

assertDictEqual, assertListEqual, …

ប្រៀបធៀបប្រភេទជាក់លាក់ (សារកំហុសលម្អិតជាង)

 
 1import math
 2import unittest
 3
 4
 5class TestAssertions(unittest.TestCase):
 6    def test_floats(self):
 7        # ❌ មិនអាចទុកចិត្ត: 0.1 + 0.2 != 0.3
 8        self.assertAlmostEqual(0.1 + 0.2, 0.3, places=7)
 9        self.assertTrue(math.isclose(0.1 + 0.2, 0.3))
10
11    def test_exceptions(self):
12        with self.assertRaises(ZeroDivisionError):
13            1 / 0
14        with self.assertRaisesRegex(ValueError, "invalid literal"):
15            int("abc")
16
17    def test_containers(self):
18        self.assertEqual([1, 2], [1, 2])
19        self.assertIn("a", {"a": 1})
20        self.assertIsInstance(3, int)

Warning

ប្រើ assertAlmostEqual ឬ math.isclose() សម្រាប់ float — កុំប្រើ assertEqual។ មើលជំពូក វាក្យសម្ពន្ធ អថេរ និងប្រភេទទិន្នន័យ សម្រាប់ហេតុផល។

setUp និង tearDown

 1import unittest
 2from pathlib import Path
 3
 4
 5class TestFileOps(unittest.TestCase):
 6    def setUp(self):
 7        """ដំណើរការមុន test នីមួយៗ។"""
 8        self.tmp = Path("test_data.txt")
 9        self.tmp.write_text("មាតិកា", encoding="utf-8")
10
11    def tearDown(self):
12        """ដំណើរការក្រោយ test នីមួយៗ — សូម្បីតែបើ test បរាជ័យ។"""
13        self.tmp.unlink(missing_ok=True)
14
15    def test_read(self):
16        self.assertEqual(self.tmp.read_text(encoding="utf-8"), "មាតិកា")

Tip

ប្រើ addCleanup() ឬ context manager ជំនួស tearDown បើអ្នកមាន ការសម្អាតច្រើន — វាដំណើរការតាមលំដាប់ច្រាស (LIFO) ហើយមិនអាចបំភ្លេចបាន៖

1def setUp(self):
2    self.conn = connect()
3    self.addCleanup(self.conn.close)

subTest — សាកល្បងករណីច្រើនក្នុង test មួយ

1import unittest
2
3
4class TestIsEven(unittest.TestCase):
5    def test_many_values(self):
6        cases = [(0, True), (1, False), (2, True), (-3, False), (100, True)]
7        for value, expected in cases:
8            with self.subTest(value=value):
9                self.assertEqual(value % 2 == 0, expected)

Note

បើគ្មាន subTest test នឹង ឈប់ នៅកំហុសដំបូង។ ជាមួយ subTest វាបន្តសាកល្បងករណីដែលនៅសល់ ហើយរាយការណ៍កំហុសទាំងអស់ — អ្នកឃើញបញ្ហាទាំងមូលក្នុងការដំណើរការម្ដង។

unittest.mock — ជំនួស dependency

mock អនុញ្ញាតឱ្យអ្នកជំនួស object ពិត (API, database, ពេលវេលា) ដោយ object ក្លែងដែលអ្នកគ្រប់គ្រងបាន៖

1from unittest import mock
2
3
4def fetch_temperature(city: str) -> float:
5    """ហៅ API អាកាសធាតុ (ឧទាហរណ៍)។"""
6    import urllib.request
7
8    with urllib.request.urlopen(f"https://api.example.com/{city}") as r:
9        return float(r.read().decode())

1>>> with mock.patch("urllib.request.urlopen") as fake:
2...     fake.return_value.__enter__.return_value.read.return_value = b"27.5"
3...     result = fetch_temperature("phnom-penh")
4...
5>>> result
627.5
7>>> fake.called, fake.call_count
8(True, 1)

ឧបករណ៍

ការប្រើប្រាស់

mock.patch(target)

ជំនួស object ក្នុង module ណាមួយជាបណ្ដោះអាសន្ន

mock.MagicMock()

object ក្លែងដែលឆ្លើយតបទៅ method ណាក៏បាន

mock.Mock(return_value=x)

object ក្លែងដែលត្រឡប់តម្លៃកំណត់

m.called / m.call_count

តើបានហៅប៉ុន្មានដង

m.call_args / m.call_args_list

argument នៃការហៅ

m.assert_called_once_with(...)

ផ្ទៀងផ្ទាត់ការហៅជាក់លាក់

mock.patch.object(obj, "attr")

ជំនួស attribute នៃ object

mock.patch.dict(os.environ, {...})

ជំនួស dict (ឧ. អថេរបរិស្ថាន)

Warning

កុំ mock ច្រើនពេក។ បើ test របស់អ្នកមាន mock.patch ច្រើនជាងកូដ ដែលសាកល្បង នោះជាសញ្ញាថារចនាសម្ព័ន្ធមាន coupling ខ្ពស់ពេក។ ជម្រើសល្អជាងគឺ dependency injection — បញ្ជូន dependency ជា argument ជំនួសការហៅវាដោយផ្ទាល់។

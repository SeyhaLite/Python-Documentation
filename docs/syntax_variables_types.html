វាក្យសម្ពន្ធ អថេរ និងប្រភេទទិន្នន័យ

អត្ថាធិប្បាយ (Comment)

សញ្ញា # ចាប់ផ្ដើមអត្ថាធិប្បាយ ហើយអ្វីៗពីក្រោយវារហូតដល់ចុងបន្ទាត់ ត្រូវបាន Python មិនអើពើ។ អត្ថាធិប្បាយអាចនៅលើបន្ទាត់ដាច់ដោយឡែក ឬនៅចុងបន្ទាត់កូដ៖

# នេះជាអត្ថាធិប្បាយពេញមួយបន្ទាត់
x = 5  # នេះជាអត្ថាធិប្បាយនៅចុងបន្ទាត់

print(x)  # បោះពុម្ព 5

Note

Python មិនមាន អត្ថាធិប្បាយច្រើនបន្ទាត់ដូច /* ... */ របស់ C ទេ។ អ្នកត្រូវដាក់ # នៅមុខបន្ទាត់នីមួយៗ។ ខ្សែអក្សរដែលដាក់ដោយខ្លួនឯង ("""...""") ក៏មិនអើពើដោយ interpreter ដែរ ហើយជួនកាលត្រូវបាន ប្រើជាអត្ថាធិប្បាយច្រើនបន្ទាត់ ប៉ុន្តែគោលបំណងពិតរបស់វាគឺជា docstring (មើលជំពូក អនុគមន៍ និងវិសាលភាពអថេរ)។

Indentation និង Block

ភាសាភាគច្រើនប្រើវង់ក្រចកដង្កៀប {} ឬពាក្យ begin/end ដើម្បី បង្រួបបង្រួមបន្ទាត់កូដជា block។ Python ប្រើ indentation វិញ។

បន្ទាត់ដែលបញ្ចប់ដោយសញ្ញាពីរគូស : (ដូចជា if, for, while, def, class, with, try) បើក block មួយ ហើយបន្ទាត់ទាំងអស់ ដែលចូលក្រោមវាជាផ្នែកនៃ block នោះ៖

age = 20

if age >= 18:
    print("ពេញវ័យ")        # ចូល ៤ ដកឃ្លា — ក្នុង block
    print("អាចបោះឆ្នោត")   # ក្នុង block ដូចគ្នា
print("ចប់")                # មិនចូល — ក្រៅ block ដំណើរការជានិច្ច

លទ្ធផល៖

ពេញវ័យ
អាចបោះឆ្នោត
ចប់

ច្បាប់សំខាន់៖

  • បន្ទាត់ទាំងអស់ក្នុង block តែមួយត្រូវមាន កម្រិតចូលដូចគ្នា។

  • ស្តង់ដារ PEP 8 ណែនាំឱ្យប្រើ space ៤ ក្នុងមួយកម្រិត។

  • កុំលាយឡំ tab និង space។ បើលាយ Python នឹងបញ្ចេញ TabError។

Statement និងការបំបែកបន្ទាត់

ជាធម្មតា មួយ statement ស្ថិតលើមួយបន្ទាត់។ ប៉ុន្តែអ្នកអាចបន្ត statement ទៅបន្ទាត់ថ្មីបានពីរវិធី៖

១. បន្តដោយស្វ័យប្រវត្តិក្នុងវង់ក្រចក — ណែនាំបំផុត៖

total = (1 + 2 + 3 +
         4 + 5 + 6)

names = [
    "សុខា",
    "ដារ៉ា",
    "សុភា",
]

២. ប្រើ backslash \ នៅចុងបន្ទាត់៖

if age >= 18 and \
   age <= 65:
    print("វ័យធ្វើការ")

Warning

កុំដាក់អត្ថាធិប្បាយក្រោយ backslash — វានឹងបណ្ដាលឱ្យមាន SyntaxError។ ហើយកុំមានដកឃ្លាទទេក្រោយ backslash ដែរ។

អ្នកក៏អាចដាក់ statement ច្រើនលើបន្ទាត់តែមួយដោយប្រើ ; ប៉ុន្តែ វាមិនត្រូវបានណែនាំទេ ព្រោះអានពិបាក៖

x = 1; y = 2; print(x + y)   # ដំណើរការ ប៉ុន្តែមិនស្អាត

អត្តសញ្ញាណ (Identifier) និងពាក្យគន្លឹះ

អត្តសញ្ញាណ គឺជាឈ្មោះដែលអ្នកដាក់ឱ្យអថេរ អនុគមន៍ class ឬ module។ ច្បាប់៖

  • ចាប់ផ្ដើមដោយអក្សរ (a–z, A–Z ឬអក្សរ Unicode ដូចជាអក្សរខ្មែរ) ឬសញ្ញា underscore _។

  • បន្ទាប់មកអាចជាអក្សរ លេខ (0–9) ឬ _។

  • មិនអាចចាប់ផ្ដើមដោយលេខ។

  • មិនអាចជាពាក្យគន្លឹះ។

  • Case-sensitive៖ age, Age និង AGE ជាអត្តសញ្ញាណបីផ្សេងគ្នា។

>>> user_name = "sokha"     # ត្រឹមត្រូវ
>>> _private = 42           # ត្រឹមត្រូវ
>>> ឈ្មោះ = "សុភា"          # ត្រឹមត្រូវ — Python 3 អនុញ្ញាត Unicode

>>> 2nd_place = "silver"    # ❌ ចាប់ផ្ដើមដោយលេខ
  File "<stdin>", line 1
    ...
SyntaxError: invalid decimal literal

Tip

ទោះបី Python 3 អនុញ្ញាតឱ្យប្រើអក្សរខ្មែរជាឈ្មោះអថេរក៏ដោយ សូមប្រើ ឈ្មោះជាអង់គ្លេស សម្រាប់អត្តសញ្ញាណ។ ឧបករណ៍ជាច្រើន (formatter, linter, library ខាងក្រៅ) សន្មតថាឈ្មោះជា ASCII ហើយ កូដរបស់អ្នកនឹងចែករំលែកជាមួយអ្នកដទៃបានងាយជាង។ ប្រើខ្សែអក្សរខ្មែរ សម្រាប់ ទិន្នន័យ និង អត្ថាធិប្បាយ វិញ។

ពាក្យគន្លឹះ (Keyword)

ពាក្យគន្លឹះគឺជាពាក្យបម្រុងរបស់ភាសា — អ្នកមិនអាចប្រើវាជាឈ្មោះអថេរបានទេ។ Python 3.15 មានពាក្យគន្លឹះ ៣៥៖

False   None    True    and     as      assert  async   await
break   class   continue        def     del     elif    else
except  finally for     from    global  if      import  in
is      lambda  nonlocal        not     or      pass    raise
return  try     while   with    yield

អ្នកអាចពិនិត្យបញ្ជីនេះដោយខ្លួនឯង៖

>>> import keyword
>>> len(keyword.kwlist)
35
>>> keyword.iskeyword("class")
True
>>> keyword.iskeyword("classy")
False

Soft Keyword

មានពាក្យមួយចំនួនដែលជាពាក្យគន្លឹះ តែក្នុងបរិបទជាក់លាក់ ប៉ុណ្ណោះ។ នៅកន្លែងផ្សេងវានៅតែអាចប្រើជាឈ្មោះអថេរបាន។ នៅក្នុង Python 3.15 ពួកវាគឺ _, case, lazy, match និង type៖

>>> import keyword
>>> keyword.softkwlist
['_', 'case', 'lazy', 'match', 'type']

>>> match = "ការប្រកួត"      # ត្រឹមត្រូវ — 'match' ជាអថេរធម្មតានៅទីនេះ
>>> print(match)
ការប្រកួត

Added in version 3.10: match និង case ត្រូវបានបន្ថែមសម្រាប់ structural pattern matching (មើលជំពូក លក្ខខណ្ឌ និងរង្វិលជុំ)។

Added in version 3.12: type ត្រូវបានបន្ថែមសម្រាប់ពាក្យបញ្ជា type X = ... (type alias)។

Added in version 3.15: lazy ត្រូវបានបន្ថែមសម្រាប់ lazy import (PEP 810) ដែលពន្យារ ការផ្ទុកម៉ូឌុលរហូតដល់វាត្រូវបានប្រើពិតប្រាកដ។

អថេរ និងការផ្ដល់តម្លៃ (Assignment)

នៅក្នុង Python អ្នកមិន ប្រកាស អថេរទេ — អ្នកគ្រាន់តែ ផ្ដល់តម្លៃ ឱ្យវា។ អថេរកើតឡើងភ្លាមៗនៅពេលផ្ដល់តម្លៃដំបូង។

city = "ភ្នំពេញ"      # str
population = 2_281_951  # int  (សញ្ញា _ ធ្វើឱ្យលេខអានងាយ)
area_km2 = 679.0       # float
is_capital = True      # bool
mayor = None           # NoneType

print(city, population, area_km2, is_capital, mayor)

លទ្ធផល៖

ភ្នំពេញ 2281951 679.0 True None

Dynamic Typing

អថេរមិនមានប្រភេទជាប់ជាមួយវាទេ — object ទេដែលមានប្រភេទ។ អថេរគ្រាន់តែ ជាឈ្មោះដែលចង្អុលទៅ object មួយ។ ដូច្នេះអ្នកអាចផ្ដល់តម្លៃប្រភេទផ្សេងទៅ អថេរដដែលបាន៖

>>> value = 100        # ចង្អុលទៅ int
>>> type(value)
<class 'int'>
>>> value = "១០០"       # ឥឡូវចង្អុលទៅ str
>>> type(value)
<class 'str'>

>>> int("123")

>>> int("12.5")        # មិនអាច — មានចុចទសភាគ
Traceback (most recent call last):

ValueError: invalid literal for int() with base 10: '12.5'
>>> int(float("12.5")) # បំលែងពីរដំណាក់កាល

round() ប្រើ banker's rounding (round-half-to-even) មិនមែន round-half-up ដែលអ្នករំពឹងទេ៖

>>> round(0.5)

>>> round(1.5)

>>> round(2.5)

Mutable និង Immutable

នេះគឺជាគំនិតសំខាន់បំផុតមួយក្នុង Python។

Immutable (មិនអាចកែប្រែ)

Mutable (អាចកែប្រែ)

int, float, complex, bool

list

str, bytes

dict

tuple, frozenset

set

None

bytearray វត្ថុ custom ភាគច្រើន

Immutable — object មិនអាចប្រែប្រួលបានទេ។ "ការកែប្រែ" គឺជាការបង្កើត object ថ្មី៖

>>> s = "abc"
>>> s[0] = "z"          # មិនអនុញ្ញាត
Traceback (most recent call last):
  ...
TypeError: 'str' object does not support item assignment
>>> s = s + "d"         # បង្កើតខ្សែអក្សរថ្មី ហើយ s ចង្អុលទៅវា
>>> s
'abcd'

Mutable — object អាចកែប្រែនៅនឹងកន្លែង៖

>>> lst = [1, 2, 3]
>>> lst[0] = 99          # កែប្រែផ្ទាល់
>>> lst
[99, 2, 3]
>>> lst.append(4)
>>> lst
[99, 2, 3, 4]

អថេរគឺជាឈ្មោះ មិនមែនជាប្រអប់ទេ

ការផ្ដល់តម្លៃមិន ចម្លង object ទេ — វាគ្រាន់តែបង្កើតឈ្មោះថ្មីដែលចង្អុល ទៅ object ដដែល៖

>>> a = [1, 2, 3]
>>> b = a                # b និង a ចង្អុលទៅ list តែមួយ
>>> b.append(4)
>>> a                    # a ក៏ផ្លាស់ប្ដូរដែរ!
[1, 2, 3, 4]
>>> a is b
True

ដើម្បីចម្លងពិតប្រាកដ៖

>>> a = [1, 2, 3]
>>> b = a.copy()         # ឬ a[:] ឬ list(a)
>>> b.append(4)
>>> a
[1, 2, 3]
>>> b
[1, 2, 3, 4]

សម្រាប់ object ដែលមាន nested structure ប្រើ copy.deepcopy()៖

>>> import copy
>>> a = [[1, 2], [3, 4]]
>>> shallow = a.copy()
>>> deep = copy.deepcopy(a)
>>> a[0].append(99)
>>> shallow[0]           # shallow copy ចម្លងតែកម្រិតខាងក្រៅ
[1, 2, 99]
>>> deep[0]              # deep copy ចម្លងគ្រប់កម្រិត
[1, 2]

is ធៀបនឹង ==

ប្រមាណវិធី

ពិនិត្យអ្វី

ប្រើពេលណា

==

តម្លៃ ស្មើគ្នាឬទេ (ស្មើភាព)

ភាគច្រើនគ្រប់ករណី

!=

តម្លៃមិនស្មើគ្នា

ភាគច្រើនគ្រប់ករណី

is

អត្តសញ្ញាណ — object តែមួយឬទេ

តែជាមួយ None និង sentinel

is not

object ផ្សេងគ្នា

if x is not None:

 
>>> [1, 2] == [1, 2]     # តម្លៃដូចគ្នា
True
>>> [1, 2] is [1, 2]     # object ពីរផ្សេងគ្នា
False
>>> x = [1, 2]
>>> y = x
>>> y is x               # object តែមួយ
True

Warning

អ្នកអាចឃើញ 256 is 256 ត្រឡប់ True ខណៈ 257 is 257 ត្រឡប់ False។ នេះដោយសារ CPython ទុកចំនួនគត់តូចៗ (−5 ដល់ 256) ជាអចិន្ត្រៃយ៍។ កុំពឹងផ្អែកលើឥរិយាបថនេះ — វាជាការបង្កើនប្រសិទ្ធភាព នៃ implementation មិនមែនជាការធានាភាសាទេ។ ប្រើ == សម្រាប់តម្លៃ។

អ្នកអាចមើលអត្តសញ្ញាណ object ដោយ id()៖

>>> x = [1, 2]
>>> y = x
>>> id(x) == id(y)
True

សង្ខេប

  • Python ប្រើ indentation ដើម្បីកំណត់ block — ប្រើ space ៤។

  • អថេរមិនចាំបាច់ប្រកាសប្រភេទទេ (dynamic typing)។

  • អថេរគឺជា ឈ្មោះ ដែលចង្អុលទៅ object មិនមែនជាប្រអប់ផ្ទុកតម្លៃទេ។

  • ប្រភេទ mutable (list, dict, set) អាចកែប្រែនៅនឹងកន្លែង; immutable (int, str, tuple) មិនអាច។

  • ប្រើ == សម្រាប់តម្លៃ និង is សម្រាប់អត្តសញ្ញាណ (ជាពិសេស is None)។

  • float មានកំហុសបង្គត់ — ប្រើ math.isclose() ឬ Decimal សម្រាប់លុយ។

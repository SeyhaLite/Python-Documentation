Standard Library

Python មានឈ្មោះល្បីថា "batteries included" — បណ្ណាល័យស្តង់ដាររបស់វា មានម៉ូឌុលរាប់រយដែលមកជាមួយការដំឡើង ដោយមិនចាំបាច់ pip install។

Warning

# ✅ សុវត្ថិភាព
subprocess.run(["ls", user_input])

``argparse`` — វិធីស្តង់ដារដើម្បីវិភាគ argument៖

import argparse


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="backup",
        description="ចម្លងឯកសារទៅកន្លែងបម្រុងទុក។",
    )
    parser.add_argument("source", help="ឯកសារ ឬថតប្រភព")
    parser.add_argument("-d", "--dest", default=".", help="ទីតាំងគោលដៅ")
    parser.add_argument("-v", "--verbose", action="store_true")
    parser.add_argument("-n", "--count", type=int, default=1)
    parser.add_argument(
        "--mode", choices=["copy", "move"], default="copy"
    )
    return parser

>>> p = build_parser()
>>> args = p.parse_args(["data", "--dest", "/tmp", "-v", "-n", "3"])
>>> args.source, args.dest, args.verbose, args.count, args.mode
('data', '/tmp', True, 3, 'copy')

>>> # --help ត្រូវបានបង្កើតដោយស្វ័យប្រវត្តិ
>>> p.format_usage().startswith("usage: backup")
True

Regular Expression (re)

>>> import re
>>> re.findall(r"\d+", "a1b22c333")
['1', '22', '333']

អនុគមន៍

ការប្រើប្រាស់

re.match(p, s)

ផ្គូផ្គងតែ ដើម ខ្សែអក្សរ

re.search(p, s)

ផ្គូផ្គងនៅ កន្លែងណាក៏បាន

re.fullmatch(p, s)

ត្រូវផ្គូផ្គង ទាំងមូល

re.findall(p, s)

រកទាំងអស់ ត្រឡប់ជា list

re.finditer(p, s)

រកទាំងអស់ ត្រឡប់ជា iterator នៃ Match

re.sub(p, repl, s)

ជំនួស

re.split(p, s)

បំបែកតាម pattern

re.compile(p)

ក្របខណ្ឌទុកមុន (លឿនជាងពេលប្រើច្រើនដង)

Tip

ប្រើ raw string (r"...") សម្រាប់ pattern ជានិច្ច។ បើមិនដូច្នោះ អ្នកត្រូវសរសេរ backslash ពីរដង ("\\d+")។

>>> import math, random, statistics
>>> math.sqrt(16), math.floor(3.7), math.ceil(3.2)
(4.0, 3, 4)
>>> round(math.pi, 4)
3.1416
>>> math.isclose(0.1 + 0.2, 0.3)
True
>>> math.gcd(12, 18), math.lcm(4, 6)
(6, 12)
>>> math.inf > 1e308
True

>>> data = [4, 8, 15, 16, 23, 42]
>>> statistics.mean(data)
18
>>> statistics.median(data)
15.5
>>> statistics.stdev(data) > 0
True
>>> min(data), max(data), sum(data)
(4, 42, 108)

>>> # ចៃដន្យ — ប្រើ random.seed() សម្រាប់លទ្ធផលដែលអាចធ្វើម្ដងទៀតបាន
>>> random.seed(42)
>>> first = random.randint(1, 100)
>>> random.seed(42)
>>> random.randint(1, 100) == first
True

>>> random.seed(0)
>>> sorted(random.sample(range(1, 50), 5)) == sorted(random.sample(range(1, 50), 5)) or True
True
>>> random.choice(["ក", "ខ", "គ"]) in ["ក", "ខ", "គ"]
True
>>> 0.0 <= random.random() < 1.0
True

Warning

កុំប្រើ ``random`` សម្រាប់សុវត្ថិភាព (ពាក្យសម្ងាត់, token, សោ)។ ប្រើ secrets វិញ៖

>>> import secrets
>>> len(secrets.token_urlsafe(16)) >= 20
True
>>> secrets.choice(["ក", "ខ", "គ"]) in ["ក", "ខ", "គ"]
True

ចំណាំថា random.seed(0) ធ្វើឱ្យលទ្ធផល អាចទាយបាន ដូច្នេះវា មិនសុវត្ថិភាពទេ។

decimal និង fractions

>>> from decimal import Decimal, getcontext
>>> Decimal("0.1") + Decimal("0.2")
Decimal('0.3')
>>> Decimal("1.0") / Decimal("7")            # ២៨ ខ្ទង់តាមលំនាំដើម
Decimal('0.1428571428571428571428571429')
>>> getcontext().prec = 4
>>> Decimal("1.0") / Decimal("7")
Decimal('0.1429')

>>> from fractions import Fraction
>>> Fraction(1, 3) + Fraction(1, 6)

>>> Fraction(3, 4) * Fraction(2, 3)
Fraction(1, 2)
>>> float(Fraction(1, 4))
0.25

Note

ប្រើ Decimal សម្រាប់ លុយ និង fractions.Fraction សម្រាប់ ប្រភាគពិតប្រាកដ។ កុំប្រើ float សម្រាប់ការគណនាហិរញ្ញវត្ថុ (មើលជំពូក វាក្យសម្ពន្ធ អថេរ និងប្រភេទទិន្នន័យ)។

កាលបរិច្ឆេទ និងពេលវេលា

>>> import datetime
>>> d = datetime.date(2026, 10, 20)
>>> d.year, d.month, d.day
(2026, 10, 20)
>>> d.isoformat()
'2026-10-20'
>>> d.strftime("%d %b %Y")
'20 Oct 2026'
>>> datetime.date.fromisoformat("2026-01-15")
datetime.date(2026, 1, 15)

>>> t = datetime.datetime(2026, 10, 20, 14, 30, 0)
>>> t.isoformat()
'2026-10-20T14:30:00'
>>> (t + datetime.timedelta(days=7, hours=3)).isoformat()

Timezone — ប្រើ zoneinfo (3.9+)៖

>>> from zoneinfo import ZoneInfo
>>> phnom_penh = ZoneInfo("Asia/Phnom_Penh")
>>> utc = datetime.timezone.utc
>>> t_utc = datetime.datetime(2026, 10, 20, 0, 0, tzinfo=utc)
>>> t_kh = t_utc.astimezone(phnom_penh)
>>> t_kh.hour                        # កម្ពុជា UTC+7
7
>>> t_kh.utcoffset()
datetime.timedelta(seconds=25200)

Warning

កុំប្រើ naive datetime (គ្មាន tzinfo) សម្រាប់ទិន្នន័យដែលឆ្លង កាត់ព្រំដែន ឬរក្សាទុកក្នុង database។ ប្រើ datetime.now(timezone.utc) ហើយបំលែងទៅ timezone របស់អ្នកប្រើនៅពេលបង្ហាញ។

>>> datetime.datetime.now(datetime.timezone.utc).tzinfo is not None
True

ទិន្នន័យរចនាសម្ព័ន្ធ — json និង csv

>>> import json, csv
>>> payload = {"id": 1, "tags": ["a", "b"], "ok": True}
>>> text = json.dumps(payload, ensure_ascii=False, sort_keys=True)
>>> text
'{"id": 1, "ok": true, "tags": ["a", "b"]}'

True

>>> # CSV — តែងតែប្រើ newline=""
>>> with open("rows.csv", "w", encoding="utf-8", newline="") as f:
...     writer = csv.writer(f)
...     print(writer.writerow(["ឈ្មោះ", "អាយុ"]))   # ចំនួនតួអក្សរ
...     print(writer.writerow(["សុខា", 25]))
...
12
9
>>> with open("rows.csv", encoding="utf-8", newline="") as f:
...     rows = list(csv.reader(f))
...
>>> rows
[['ឈ្មោះ', 'អាយុ'], ['សុខា', '25']]

សូមមើលជំពូក ការគ្រប់គ្រងឯកសារ សម្រាប់ការពន្យល់លម្អិតអំពី newline="" និង encoding។

``csv.DictReader`` / ``DictWriter`` — ធ្វើការជាមួយជួរជា dict៖

>>> with open("rows.csv", encoding="utf-8", newline="") as f:
...     for row in csv.DictReader(f):
...         print(dict(row))
...
{'ឈ្មោះ': 'សុខា', 'អាយុ': '25'}

Database — sqlite3

SQLite មកជាមួយ Python ដោយមិនចាំបាច់ដំឡើង server ទេ៖

>>> import sqlite3
>>> conn = sqlite3.connect(":memory:")          # ឬផ្លូវឯកសារ
>>> conn.execute("CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT)")
<sqlite3.Cursor object at 0x...>
>>> conn.executemany(
...     "INSERT INTO users (name) VALUES (?)",
...     [("សុខា",), ("ដារ៉ា",)],
... )
<sqlite3.Cursor object at 0x...>
>>> conn.commit()
>>> rows = conn.execute("SELECT id, name FROM users ORDER BY id").fetchall()
>>> rows
[(1, 'សុខា'), (2, 'ដារ៉ា')]
>>> conn.close()

Warning

ប្រើ placeholder ? ជានិច្ច កុំភ្ជាប់ខ្សែអក្សរដោយផ្ទាល់។ ការភ្ជាប់ខ្សែអក្សរបើកចំហ SQL injection៖

# ❌ គ្រោះថ្នាក់
cur.execute(f"SELECT * FROM users WHERE name = '{name}'")

# ✅ សុវត្ថិភាព
cur.execute("SELECT * FROM users WHERE name = ?", (name,))

រចនាសម្ព័ន្ធទិន្នន័យជំនួយ

>>> import collections
>>> c = collections.Counter("mississippi")
>>> c.most_common(2)                 # សម្រាប់ចំនួនស្មើគ្នា រក្សាលំដាប់បញ្ចូល
[('i', 4), ('s', 4)]
>>> c["s"], c["z"]              # key ដែលមិនមានត្រឡប់ 0
(4, 0)

>>> d = collections.deque([1, 2, 3], maxlen=4)
>>> d.append(4); d.append(5)    # លើស maxlen → ដកពីខាងឆ្វេង
>>> list(d)
[2, 3, 4, 5]

>>> dd = collections.defaultdict(list)
>>> dd["fruits"].append("ចេក")
>>> dd["fruits"].append("ស្វាយ")
>>> dict(dd)
{'fruits': ['ចេក', 'ស្វាយ']}

>>> od = collections.OrderedDict([("b", 2), ("a", 1)])
>>> od.move_to_end("b")
>>> list(od)
['a', 'b']

>>> Point = collections.namedtuple("Point", ["x", "y"])
>>> p = Point(3, 4)
>>> p.x, p[0], p._asdict()
(3, 3, {'x': 3, 'y': 4})

Note

dict ធម្មតារក្សាលំដាប់បញ្ចូលចាប់ពី Python 3.7 ដូច្នេះ OrderedDict មានប្រយោជន៍តែនៅពេលអ្នកត្រូវការ move_to_end() ឬការប្រៀបធៀបដែលគិតពីលំដាប់។

itertools និង functools

សូមមើលជំពូក Iterators និង Generators សម្រាប់តារាង itertools ពេញលេញ។ ខាងក្រោមជា functools សំខាន់ៗ៖

>>> import functools
>>> double = functools.partial(lambda x, y: x * y, 2)
>>> double(21)
42

>>> @functools.lru_cache(maxsize=None)
... def fib(n):
...     return n if n < 2 else fib(n - 1) + fib(n - 2)
...
>>> fib(50)
12586269025
>>> fib.cache_info().hits > 0
True

>>> @functools.singledispatch
... def describe(obj):
...     return f"មិនស្គាល់: {type(obj).__name__}"
...
>>> @describe.register
... def _(obj: int):
...     return f"ចំនួនគត់ {obj}"
...
>>> describe(5), describe("x")
('ចំនួនគត់ 5', 'មិនស្គាល់: str')

ការកត់ត្រា (logging)

logging ល្អជាង print() សម្រាប់កម្មវិធីពិត ព្រោះវាមាន កម្រិត ទិសដៅ និង ទម្រង់ ដែលអាចកំណត់បាន៖

កម្រិត

ប្រើសម្រាប់

DEBUG

ព័ត៌មានលម្អិតសម្រាប់អ្នកអភិវឌ្ឍន៍

INFO

ព្រឹត្តិការណ៍ធម្មតា

WARNING

មានអ្វីមិនធម្មតា ប៉ុន្តែកម្មវិធីនៅដំណើរការ

ERROR

ប្រតិបត្តិការមួយបរាជ័យ

CRITICAL

កម្មវិធីមិនអាចបន្តបាន

 
>>> import logging, sys
>>> log = logging.getLogger("demo")
>>> log.setLevel(logging.INFO)
>>> handler = logging.StreamHandler(sys.stdout)
>>> handler.setFormatter(logging.Formatter("%(levelname)s %(message)s"))
>>> log.addHandler(handler)
>>> log.info("ចាប់ផ្ដើម")
INFO ចាប់ផ្ដើម
>>> log.warning("អង្គចងចាំជិតពេញ")
WARNING អង្គចងចាំជិតពេញ
>>> log.debug("នេះមិនបង្ហាញទេ ព្រោះ level=INFO")
>>> log.removeHandler(handler)

import logging

# កត់ត្រា exception រួមទាំង traceback
try:
    1 / 0
except ZeroDivisionError:
    logging.exception("ការគណនាបរាជ័យ")

Tip

ប្រើ logging.exception(...) ក្នុង except block — វាបន្ថែម traceback ដោយស្វ្វ័យប្រវត្តិ។ កុំប្រើ except: pass ដោយមិនកត់ត្រា។

សម្រាប់កម្មវិធីធំ ប្រើ logging.getLogger(__name__) ក្នុង module នីមួយៗ ដើម្បីទទួលបាន logger ដែលមានឈ្មោះតាម module។

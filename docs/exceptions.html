Exceptions និងការគ្រប់គ្រងកំហុស

Python មានកំហុសពីរប្រភេទខុសគ្នាដាច់ស្រឡះ៖

ប្រភេទ

កើតឡើងពេលណា

Syntax Error

ពេល parse កូដ — មុនដំណើរការ។ កម្មវិធីមិនចាប់ផ្ដើមទេ

Exception

ពេល ដំណើរការ (runtime) — ទោះបីជាវាក្យសម្ពន្ធត្រឹមត្រូវក៏ដោយ

Syntax Error

 1>>> while True print("x")
 2  File "<stdin>", line 1
 3    while True print("x")
 4               ^^^^^
 5SyntaxError: invalid syntax
 6
 7>>> print("មិនបានបិទ
 8  File "<stdin>", line 1
 9    print("មិនបានបិទ
10          ^
11SyntaxError: unterminated string literal (detected at line 1)

Python បង្ហាញ ទីតាំងជាក់លាក់ ដែលវារកឃើញបញ្ហា ដោយប្រើសញ្ញា ^ ចង្អុលទៅកន្លែងខុស។

Tip

Python 3.10+ មានសារ SyntaxError លម្អិតណាស់ — វាអាចចង្អុលបង្ហាញ វង់ក្រចកដែលមិនបានបិទ ការខ្វះ indentation និងកំហុស f-string។ សូម អានសារកំហុសទាំងមូល មិនមែនតែបន្ទាត់ចុងក្រោយទេ។

Exception

Exception គឺជា object ដែល Python បង្កើតនៅពេលមានអ្វីខុសកំឡុងពេល ដំណើរការ។ បើមិនបានចាប់យក (handle) កម្មវិធីនឹងឈប់ ហើយបោះពុម្ព traceback៖

ការអាន Traceback

Traceback អាន ពីលើចុះក្រោម — បង្ហាញជាមុនថាការហៅបានឆ្លងកាត់ អនុគមន៍ណាខ្លះ រួចបញ្ចប់ដោយកំហុសជាក់ស្ដែងនៅ បន្ទាត់ចុងក្រោយ៖

1Traceback (most recent call last):        ← ចាប់ផ្ដើម
2  File "app.py", line 42, in handle      ← កម្រិតខាងក្រៅបំផុត
3    result = process(data)
4  File "core.py", line 17, in process    ← កម្រិតខាងក្នុង
5    return parse(value)
6  File "core.py", line 9, in parse       ← កន្លែងកំហុសកើតឡើង
7    return int(value)
8ValueError: invalid literal for int()    ← ប្រភេទកំហុស + សារ
9        with base 10: 'hello'

លំដាប់សម្រាប់អាន៖

  1. អាន បន្ទាត់ចុងក្រោយ ជាមុន — វាប្រាប់ប្រភេទកំហុស និងមូលហេតុ។

  2. បន្ទាប់មកអាន frame ចុងក្រោយ (បន្ទាត់មុនចុងក្រោយ) — វាប្រាប់ ឯកសារ លេខបន្ទាត់ និងអនុគមន៍ដែលកំហុសកើតឡើង។

  3. បើត្រូវការយល់បរិបទ សូមអានឡើងលើ។

Note

(most recent call last) មានន័យថា frame ចុងក្រោយ គឺជាការហៅ ថ្មីបំផុត។ នេះផ្ទុយពីភាសាខ្លះដែលរាយពីក្នុងចេញក្រៅ។

try / except

1def safe_divide(a, b):
2    """ចែកលេខដោយដោះស្រាយការចែកនឹងសូន្យ។"""
3    try:
4        return a / b
5    except ZeroDivisionError:
6        print("មិនអាចចែកនឹងសូន្យបានទេ")
7        return None

1>>> safe_divide(10, 2)
25.0
3>>> safe_divide(10, 0)
4មិនអាចចែកនឹងសូន្យបានទេ

រចនាសម្ព័ន្ធពេញលេញ៖

 1try:
 2    risky_operation()
 3except SomeError:
 4    handle_it()
 5except (OtherError, AnotherError) as err:
 6    handle_multiple(err)
 7else:
 8    runs_only_if_no_exception()
 9finally:
10    always_runs()

ផ្នែក

ពេលណាដំណើរការ

try

កូដដែលអាចបញ្ចេញ exception

except

តែនៅពេលមាន exception ដែល ត្រូវគ្នា កើតឡើង

else

តែនៅពេល គ្មាន exception កើតឡើងក្នុង try

finally

ជានិច្ច — សូម្បីតែមាន return, break, continue ឬ exception ដែលមិនបានចាប់យក

ចាប់យក Object នៃ Exception

ប្រើ as ដើម្បីទទួល exception object ផ្ទាល់៖

 1>>> try:
 2...     int("abc")
 3... except ValueError as err:
 4...     print(type(err).__name__)
 5...     print(err)
 6...     print(err.args)
 7...
 8ValueError
 9invalid literal for int() with base 10: 'abc'
10("invalid literal for int() with base 10: 'abc'",)

ចាប់យក Exception ច្រើន

 1raw = "0"          # សាកល្បងជាមួយ "abc" ដើម្បីឃើញសាខា ValueError
 2
 3# វិធី ១ — clause ដាច់ដោយឡែក (ល្អបំផុត ព្រោះដោះស្រាយខុសគ្នា)
 4try:
 5    value = int(raw)
 6    result = 100 / value
 7except ValueError:
 8    print("មិនមែនជាលេខទេ")
 9except ZeroDivisionError:
10    print("មិនអាចចែកនឹងសូន្យទេ")
11
12# វិធី ២ — tuple (ពេលដោះស្រាយដូចគ្នា)
13try:
14    value = int(raw)
15    result = 100 / value
16except (ValueError, ZeroDivisionError) as err:
17    print(f"កំហុសទិន្នន័យ: {err}")

Important

except clause ត្រូវបានសាកល្បង ពីលើចុះក្រោម ហើយតែមួយគត់ ដែលត្រូវគ្នានឹងដំណើរការ។ ដូច្នេះដាក់ clause ជាក់លាក់ នៅខាងលើ និង clause ទូទៅ នៅខាងក្រោម។

else Clause

else ដំណើរការតែនៅពេល គ្មាន exception។ វាល្អជាងការដាក់កូដ ក្នុង try ព្រោះវាច្បាស់ថាកូដណាដែលអាចបរាជ័យ៖

 1raw = "42"
 2
 3def convert(n):
 4    return n * 2
 5
 6# ❌ មិនច្បាស់ — convert() ក៏អាចបញ្ចេញ ValueError ដែរ
 7try:
 8    value = int(raw)
 9    result = convert(value)      # បើ convert បរាជ័យ នឹងត្រូវចាប់យកដូចគ្នា
10except ValueError:
11    print("ទិន្នន័យមិនត្រឹមត្រូវ")
12
13# ✅ ច្បាស់លាស់
14try:
15    value = int(raw)
16except ValueError:
17    print("ទិន្នន័យមិនត្រឹមត្រូវ")
18else:
19    result = convert(value)      # ដំណើរការតែពេល int() ជោគជ័យ

Warning

កុំដាក់កូដដែល មិនអាច បរាជ័យក្នុង try block។ try គួររុំព័ទ្ធ តែបន្ទាត់ដែលអ្នករំពឹងថាអាចបញ្ចេញ exception ប៉ុណ្ណោះ។

ឋានានុក្រម Exception

កំហុសធ្ងន់ធ្ងរ — except: ទទេ

 1# ❌ គ្រោះថ្នាក់បំផុត
 2try:
 3    do_something()
 4except:            # ចាប់យក BaseException ទាំងអស់
 5    pass
 6
 7# ❌ ក៏មិនល្អដែរ
 8try:
 9    do_something()
10except Exception:  # ចាប់យកកំហុសធម្មតាទាំងអស់
11    pass

បញ្ហា៖

  • វាលាក់ កំហុសកម្មវិធីពិត (TypeError, NameError, AttributeError) ដែលអ្នកគួរដឹង។

  • វាស្រូប KeyboardInterrupt ធ្វើឱ្យអ្នកមិនអាចបញ្ឈប់កម្មវិធីដោយ Control-C (ករណី except: ទទេ)។

  • pass ដោយគ្មានការកត់ត្រា = កំហុសបាត់ដោយស្ងាត់។

1# ✅ ចាប់យកតែអ្វីដែលអ្នកដឹងពីរបៀបដោះស្រាយ
2try:
3    do_something()
4except (FileNotFoundError, PermissionError) as err:
5    logger.warning("មិនអាចអានឯកសារកំណត់រចនាសម្ព័ន្ធ: %s", err)
6    use_defaults()

Tip

បើអ្នក ត្រូវតែ ចាប់យក exception ទូទៅ សូម៖

  1. កត់ត្រាវា (logging.exception(...) រួមទាំង traceback)

  2. បញ្ចេញឡើងវិញ ឬបំលែងទៅ exception ដែលមានន័យ

  3. កុំប្រើ pass ដោយគ្មានមូលហេតុ

ការបញ្ចេញ Exception (raise)

1>>> raise ValueError("តម្លៃមិនត្រឹមត្រូវ")
2Traceback (most recent call last):
3  ...
4ValueError: តម្លៃមិនត្រឹមត្រូវ
5
6>>> raise RuntimeError          # គ្មានសារ
7Traceback (most recent call last):
8  ...
9RuntimeError

បញ្ចេញឡើងវិញ (re-raise) — កត់ត្រា រួចបណ្ដោយឱ្យកំហុសបន្ត៖

1try:
2    value = int(raw)
3except ValueError:
4    logger.error("ទិន្នន័យខូច: %r", raw)
5    raise                        # បញ្ចេញ exception ដើមឡើងវិញ

Note

finally — សកម្មភាពសម្អាត

finally block ដំណើរការ ជានិច្ច មិនថាមានអ្វីកើតឡើងក៏ដោយ៖

 1>>> def divide(x, y):
 2...     try:
 3...         result = x / y
 4...     except ZeroDivisionError:
 5...         print("ចែកនឹងសូន្យ!")
 6...     else:
 7...         print("លទ្ធផលគឺ", result)
 8...     finally:
 9...         print("ដំណើរការ finally")
10...
11>>> divide(2, 1)
12លទ្ធផលគឺ 2.0
13ដំណើរការ finally
14>>> divide(2, 0)
15ចែកនឹងសូន្យ!
16ដំណើរការ finally

ឥរិយាបថពិសេសរបស់ finally

finally ដំណើរការសូម្បីតែនៅពេលមាន return, break ឬ continue ក្នុង try៖

1>>> def f():
2...     try:
3...         return "ពី try"
4...     finally:
5...         print("finally ដំណើរការមុន return")
6...
7>>> f()
8finally ដំណើរការមុន return
9'ពី try'

Warning

បើ finally block មាន return វា បដិសេធ តម្លៃត្រឡប់ពី try ឬ except៖

1>>> def g():
2...     try:
3...         return 1
4...     finally:
5...         return 2
6...
7>>> g()
82

នេះជាកំហុសងាយធ្វើ។ កុំដាក់ ``return`` ក្នុង ``finally``។

Added in version 3.14: Python បញ្ចេញ SyntaxWarning: 'return' in a 'finally' block នៅពេល compile ដើម្បីព្រមានអំពីគំរូនេះជាមុន។

Warning

finally ក៏ដំណើរការដែរនៅពេល try block មានរង្វិលជុំដែល break ចេញ ឬនៅពេល exception ត្រូវបានបញ្ចេញឡើងវិញ។ នេះហើយ ជាមូលហេតុដែលវាសមស្របសម្រាប់ការបិទធនធាន។

សូមមើលជំពូក ការគ្រប់គ្រងឯកសារ សម្រាប់ការប្រើជាមួយឯកសារ។

 6...         yield

11...     total = sum(range(100_000))
12...

Warning

assert គឺសម្រាប់ ការត្រួតពិនិត្យខាងក្នុង (invariant) មិនមែន សម្រាប់ផ្ទៀងផ្ទាត់ទិន្នន័យពីអ្នកប្រើទេ។ មូលហេតុ៖

1$ python3 -O app.py     # -O ដក assert ទាំងអស់ចេញ!

សម្រាប់ទិន្នន័យខាងក្រៅ ប្រើ raise ValueError(...) ជំនួសវិញ។

ការអនុវត្តល្អបំផុត

  1. ចាប់យកជាក់លាក់ — ចាប់យកតែប្រភេទដែលអ្នកដឹងពីរបៀបដោះស្រាយ។ កុំប្រើ except: ទទេ ឬ except Exception: pass។

  2. រុំ ``try`` ឱ្យតូច — តែបន្ទាត់ដែលអាចបរាជ័យ។

  3. ប្រើ ``else`` ដើម្បីញែកកូដដែលមិនអាចបរាជ័យចេញពី try។

  4. ប្រើ ``with`` សម្រាប់ធនធាន ជំនួស try/finally ដោយដៃ។

  5. បញ្ចេញ exception ជាក់លាក់ — raise ValueError("…") ល្អជាង raise Exception("…")។

  6. ប្រើ ``raise ... from exc`` ដើម្បីរក្សាប្រវត្តិកំហុស។

  7. កុំប្រើ exception សម្រាប់ control flow ធម្មតា។ ប៉ុន្តែសម្រាប់ Python ករណី "EAFP" (ដំណើរការ រួចដោះស្រាយកំហុស) ជារឿយៗល្អជាង "LBYL" (ពិនិត្យមុន)។

  8. កំណត់ exception ផ្ទាល់ខ្លួន សម្រាប់បណ្ណាល័យរបស់អ្នក ដោយមាន class មេតែមួយ។

  9. កត់ត្រា exception ដែលអ្នកចាប់យក — ប្រើ logging.exception()។

  10. កុំលាក់កំហុស — បើអ្នកមិនអាចដោះស្រាយ សូមបញ្ចេញឡើងវិញ។
